CREATE TABLE JAVA_USUARIO (
    ID_USUARIO NUMBER GENERATED ALWAYS AS IDENTITY,
    NOME VARCHAR2(100) NOT NULL,
    DATA_NASCIMENTO DATE NOT NULL,
    NUMERO_CARTEIRINHA VARCHAR2(20) NOT NULL,
    EMAIL VARCHAR2(100) NOT NULL,
    SENHA VARCHAR2(50) NOT NULL,

    -- Declarando as constraints
    CONSTRAINT PK_USUARIO PRIMARY KEY (ID_USUARIO),
    CONSTRAINT UN_CARTEIRINHA UNIQUE (NUMERO_CARTEIRINHA),
    CONSTRAINT UN_EMAIL UNIQUE (EMAIL)
);

CREATE TABLE JAVA_LEMBRETE_MEDICAMENTO (
    ID_LEMBRETE NUMBER GENERATED ALWAYS AS IDENTITY,
    ID_USUARIO NUMBER NOT NULL,
    NOME_REMEDIO VARCHAR2(100) NOT NULL,
    DOSAGEM VARCHAR2(50) NOT NULL,
    HORARIO_TOMAR TIMESTAMP NOT NULL,
    STATUS_TOMOU CHAR(1) NOT NULL,

    -- Declarando as constraints
    CONSTRAINT PK_LEMBRETE PRIMARY KEY (ID_LEMBRETE),
    CONSTRAINT FK_LEMBRETE_USUARIO FOREIGN KEY (ID_USUARIO) REFERENCES JAVA_USUARIO (ID_USUARIO),
    CONSTRAINT CK_STATUS_TOMOU CHECK (STATUS_TOMOU IN ('S', 'N'))
);

CREATE TABLE JAVA_METRICA_SAUDE (
    ID_METRICA NUMBER GENERATED ALWAYS AS IDENTITY,
    ID_USUARIO NUMBER NOT NULL,
    BATIMENTOS_BPM NUMBER,
    PRESSAO_SISTOLICA NUMBER,
    PRESSAO_DIASTOLICA NUMBER,
    QTD_PASSOS NUMBER,
    CALORIAS_GASTAS NUMBER,
    DATA_HORA_REGISTRO TIMESTAMP NOT NULL,

    -- Declarando as constraints
    CONSTRAINT PK_METRICA PRIMARY KEY (ID_METRICA),
    CONSTRAINT FK_METRICA_USUARIO FOREIGN KEY (ID_USUARIO) REFERENCES JAVA_USUARIO (ID_USUARIO)
);

CREATE TABLE JAVA_AGENDAMENTO (
    ID_AGENDAMENTO NUMBER GENERATED ALWAYS AS IDENTITY,
    ID_USUARIO NUMBER NOT NULL,
    TIPO VARCHAR2(20) NOT NULL,
    MEDICO_ESPECIALIDADE VARCHAR2(100) NOT NULL,
    DATA_HORA_AGENDADA TIMESTAMP NOT NULL,

    -- Declarando as constraints
    CONSTRAINT PK_AGENDAMENTO PRIMARY KEY (ID_AGENDAMENTO),
    CONSTRAINT FK_AGENDAMENTO_USUARIO FOREIGN KEY (ID_USUARIO) REFERENCES JAVA_USUARIO (ID_USUARIO),
    CONSTRAINT CK_TIPO_AGENDAMENTO CHECK (TIPO IN ('Consulta', 'Exame'))
);
